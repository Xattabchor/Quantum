lvl = level.txt
set = setting.txt
wpn = weapon.txt
lng = lang.txt (english.txt/russian.txt)
itm = items.txt

def - Стандартное значение
nec - Опцию необходимо указать, иначе движок не будет работать

int - Целочисленное значение
float - Число с плавающей запятой
bol - Правда или ложь (0 или 1)

set/lvl SPAWN_ONE_ENEMY - (bol) (def 0) боты в устаревшем enemies= по стандарту будут спавниться только один раз 

set/lvl BOTS_CLEVER_PATHFINDING - (bol) (def 1) боты будут проверять видимость игрока при помощи пускания луча, иначе боты будут видеть игрока только в своей комнате

set/lvl OBSERVER - (bol) (def 0) Включает режим слежения за npc

set/lvl WORLD_SCALE - (float) (def 1.0) Масштаб карты

set/lvl FLOOR_OFFSETSZ (int) (def 0) Смещение пола и потолка по глубине камеры

aa a a aaa

set/lvl UPDATE_ONLY_NEAR - (bol) (def 0) отключает просчитывание далёких от игрокуа объектов

set/lvl UPDATE_ONLY_NEAR_PHYSICS - (bol) (def 0) отключает физические столкновения дальних друг от друга объектов
Не рекомендуется использовать так как для проверка проводится отдельно от физики
Удобно использовать на картах с комнатах находящихся друг в друге (прим. уровень 39)
